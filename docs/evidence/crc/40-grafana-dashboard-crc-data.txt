# The load Job
# Bounded NFS load through the IPsec tunnel: writes and reads back 8 MiB with direct I/O (so every
# byte crosses NFS, not the page cache), overwriting one file, for 25 minutes; deletes the file at the end.
apiVersion: batch/v1
kind: Job
metadata:
  name: nas-load
  namespace: ipsec-nas-demo
  labels:
    app: nas-load
spec:
  backoffLimit: 0
  activeDeadlineSeconds: 1800
  ttlSecondsAfterFinished: 3600
  template:
    metadata:
      labels:
        app: nas-load
    spec:
      restartPolicy: Never
      securityContext:
        runAsNonRoot: true
        seccompProfile:
          type: RuntimeDefault
      containers:
      - name: load
        image: registry.access.redhat.com/ubi9/python-312:latest
        command: ["/bin/bash", "-c"]
        args:
        - |
          set -uo pipefail
          f=/data/ipsec-nas-demo/load.bin
          trap 'rm -f "${f}"; echo "$(date -u +%FT%TZ) removed ${f}"' EXIT
          end=$(( $(date +%s) + 1500 ))
          n=0
          while [ "$(date +%s)" -lt "${end}" ]; do
            n=$((n + 1))
            dd if=/dev/urandom of="${f}" bs=1M count=8 oflag=direct status=none
            dd if="${f}" of=/dev/null bs=1M iflag=direct status=none
            [ $((n % 20)) -eq 0 ] && echo "$(date -u +%FT%TZ) round ${n}: 8 MiB written and read back"
            sleep 2
          done
        securityContext:
          allowPrivilegeEscalation: false
          capabilities:
            drop: ["ALL"]
        resources:
          requests: {cpu: 50m, memory: 32Mi}
          limits: {memory: 64Mi}
        volumeMounts:
        - name: nas
          mountPath: /data
      volumes:
      - name: nas
        persistentVolumeClaim:
          claimName: nas-data
